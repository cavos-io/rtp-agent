type evidenceAssistant struct {
	acceptanceEcho
	session *agent.AgentSession
	replies atomic.Int64
}

func (a *evidenceAssistant) Start(_ context.Context, s *agent.AgentSession) error {
	a.session = s
	return nil
}
func (a *evidenceAssistant) OnSpeechScheduled(_ context.Context, h *agent.SpeechHandle) {
	a.replies.Add(1)
	text := "synthetic reply: " + h.Generation.UserMessage.TextContent()
	a.session.EmitAgentOutputTranscribed(agent.AgentOutputTranscribedEvent{Transcript: text, IsFinal: true})
	h.MarkDone()
}

type observedEvidence struct {
	Sender string         `json:"sender"`
	Kind   string         `json:"kind"`
	Name   string         `json:"name"`
	Text   string         `json:"text,omitempty"`
	Data   map[string]any `json:"data,omitempty"`
}

func TestLocalNativeEvidenceDelivery(t *testing.T) {
	url, lib, out := os.Getenv("NATIVE_ACCEPTANCE_URL"), os.Getenv("LIVEKIT_LIB_PATH"), os.Getenv("NATIVE_ACCEPTANCE_OUT")
	if url != "ws://127.0.0.1:17880" || lib == "" || out == "" {
		t.Fatal("isolated URL, FFI and output required")
	}
	if err := os.Mkdir(out, 0755); err != nil {
		t.Fatal(err)
	}
	log, err := os.Create(filepath.Join(out, "events.jsonl"))
	if err != nil {
		t.Fatal(err)
	}
	defer log.Close()
	name := fmt.Sprintf("native-evidence-%d", time.Now().UnixNano())
	runID := "local-evidence"
	pub := startProcessPublisher(t, url, name, 600, 1000)
	defer pub.finish()
	received := make(chan observedEvidence, 128)
	cb := lk.NewRoomCallback()
	cb.OnDataReceived = func(data []byte, params lk.DataReceiveParams) {
		if params.Topic != "agent.events" {
			return
		}
		var ev observedEvidence
		if err := json.Unmarshal(data, &ev); err != nil {
			t.Error(err)
			return
		}
		ev.Sender = params.SenderIdentity
		received <- ev
	}
	observer, err := lk.ConnectToRoom(url, lk.ConnectInfo{APIKey: localKey, APISecret: localSecret, RoomName: name, ParticipantIdentity: "observer"}, cb)
	if err != nil {
		t.Fatal(err)
	}
	defer observer.Disconnect()
	wi.AgentNumIdleProcesses = 0
	srv := wi.NewAgentServerRuntime(wi.AgentServerRuntimeInput{AgentName: "evidence-local", LiveKitURL: url, LiveKitAPIKey: localKey, LiveKitAPISecret: localSecret}).Server
	cfg := &configmodel.ConfigModel{LiveKitURL: url, LiveKitAPIKey: localKey, LiveKitAPISecret: localSecret, NativeInProcess: true, NativeLibrary: lib}
	assistant := &evidenceAssistant{}
	jobs := make(chan *worker.JobContext, 1)
	failures := make(chan error, 16)
	if err := srv.RTCSession(func(job *worker.JobContext) error {
		session := agent.NewAgentSession(agent.NewAgent("synthetic"), nil, agent.AgentSessionOptions{})
		session.Assistant = assistant
		native := newNativeTransport(job, session, cfg, lib, nil)
		evidence := wi.NewTestEvidencePublisher(&meta.AutomatedTest{ExecutionID: runID, TesterIdentity: "caller"}, job.Room, func(err error) { failures <- err })
		if err := evidence.UseDataPublisher(native); err != nil {
			return err
		}
		detach := evidence.Attach(session)
		defer detach()
		job.SetSessionDirectory(filepath.Join(out, "session"))
		if err := job.StartSession(context.Background(), session, worker.StartSessionOptions{Transport: native, RoomOptions: worker.RoomOptions{DisablePreConnectAudio: true, ParticipantIdentity: "caller", TranscriptionNextOutput: evidence, TextInputCallback: wi.NewAgentTestTextInputCallback(evidence)}}); err != nil {
			return err
		}
		jobs <- job
		evidence.Ready()
		<-job.ShutdownDone()
		return nil
	}, nil, nil); err != nil {
		t.Fatal(err)
	}
	done := make(chan error, 1)
	go func() {
		done <- srv.ExecuteLocalJobWithOptions(context.Background(), name, "native-agent", worker.LocalJobOptions{FakeJob: false, RoomInfo: &lkproto.Room{Name: name}})
	}()
	var job *worker.JobContext
	select {
	case job = <-jobs:
	case err := <-done:
		t.Fatalf("startup: %v", err)
	case <-time.After(25 * time.Second):
		t.Fatal("startup timeout")
	}
	defer func() {
		job.Shutdown("evidence test finished")
		select {
		case err := <-done:
			if err != nil {
				t.Error(err)
			}
		case <-time.After(10 * time.Second):
			t.Error("job cleanup timeout")
		}
		if err := nativeroom.CloseInProcessRuntime(context.Background()); err != nil {
			t.Error(err)
		}
	}()
	next := func(kind, name string, turn int) observedEvidence {
		t.Helper()
		select {
		case ev := <-received:
			if err := json.NewEncoder(log).Encode(ev); err != nil {
				t.Fatal(err)
			}
			if ev.Sender != "native-agent" || ev.Kind != kind || ev.Name != name {
				t.Fatalf("unexpected event: %+v", ev)
			}
			if ev.Data["run_id"] != runID {
				t.Fatalf("wrong run: %+v", ev)
			}
			if turn > 0 && (ev.Data["message_id"] != fmt.Sprintf("%s/turn/%d", runID, turn) || ev.Data["turn"] != float64(turn)) {
				t.Fatalf("wrong correlation: %+v", ev)
			}
			return ev
		case err := <-failures:
			t.Fatalf("publication: %v", err)
		case <-time.After(8 * time.Second):
			t.Fatalf("missing %s/%s turn %d", kind, name, turn)
		}
		return observedEvidence{}
	}
	next("status", "ready_for_tester", 0)
	send := func(turn int, text string) {
		pub.room.LocalParticipant.SendText(text, lk.StreamTextOptions{Topic: "lk.chat", Attributes: map[string]string{"run_id": runID, "message_id": fmt.Sprintf("%s/turn/%d", runID, turn), "turn": fmt.Sprint(turn)}})
	}
	for turn := 1; turn <= 2; turn++ {
		send(turn, fmt.Sprintf("input %d", turn))
		next("ack", "tester_input", turn)
		ev := next("transcript", "agent", turn)
		if ev.Text != fmt.Sprintf("synthetic reply: input %d", turn) {
			t.Fatalf("wrong reply %q", ev.Text)
		}
	}
	send(2, "duplicate input")
	next("ack", "tester_input", 2)
	if ev := next("transcript", "agent", 2); ev.Text != "synthetic reply: input 2" {
		t.Fatalf("duplicate not replayed: %+v", ev)
	}
	if n := assistant.replies.Load(); n != 2 {
		t.Fatalf("reply generations = %d, want 2", n)
	}
	select {
	case ev := <-received:
		t.Fatalf("unexpected extra event: %+v", ev)
	case <-time.After(500 * time.Millisecond):
	}
	t.Log("PASS: native agent identity, ready, two correlated chat turns, duplicate replay without regeneration")
}
