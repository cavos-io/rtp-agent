const localKey = "devkey"

type acceptanceEcho struct {
	mu                      sync.Mutex
	publish                 func(context.Context, *audio.AudioFrame) error
	frames, wrong, failures atomic.Int64
	hz, other               float64
}

func (a *acceptanceEcho) Start(context.Context, *agent.AgentSession) error { return nil }
func (a *acceptanceEcho) SetPublishAudio(p func(context.Context, *audio.AudioFrame) error) {
	a.mu.Lock()
	a.publish = p
	a.mu.Unlock()
}
func (a *acceptanceEcho) OnAudioFrame(ctx context.Context, f *audio.AudioFrame) {
	pcm := make([]int16, len(f.Data)/2)
	for i := range pcm {
		pcm[i] = int16(binary.LittleEndian.Uint16(f.Data[i*2:]))
	}
	own, other := processAmplitude(pcm, int(f.SampleRate), a.hz), processAmplitude(pcm, int(f.SampleRate), a.other)
	if other > 1000 && other > own*3 {
		a.wrong.Add(1)
	}
	a.frames.Add(1)
	a.mu.Lock()
	p := a.publish
	a.mu.Unlock()
	if p != nil {
		if err := p(ctx, f); err != nil && ctx.Err() == nil {
			a.failures.Add(1)
		}
	}
}

func TestLocalNativeActiveDrainAcceptance(t *testing.T) {
	url, out, lib := os.Getenv("NATIVE_ACCEPTANCE_URL"), os.Getenv("NATIVE_ACCEPTANCE_OUT"), os.Getenv("LIVEKIT_LIB_PATH")
	if url != "ws://127.0.0.1:17880" || out == "" || lib == "" {
		t.Fatal("isolated local URL, output and FFI required")
	}
	if err := os.Mkdir(out, 0755); err != nil {
		t.Fatal(err)
	}
	file, err := os.Create(filepath.Join(out, "summary.jsonl"))
	if err != nil {
		t.Fatal(err)
	}
	defer file.Close()
	var emitMu sync.Mutex
	emit := func(v any) {
		emitMu.Lock()
		defer emitMu.Unlock()
		if err := json.NewEncoder(file).Encode(v); err != nil {
			t.Error(err)
		}
	}
	// Use the downstream server factory, native factory and hold adapter. External
	// dispatch/config/provider/storage services are intentionally outside this fixture.
	workeriface.AgentNumIdleProcesses = 0
	srv := workeriface.NewAgentServerRuntime(workeriface.AgentServerRuntimeInput{AgentName: "native-local-acceptance", LiveKitURL: url, LiveKitAPIKey: localKey, LiveKitAPISecret: localSecret}).Server
	srv.Options.DrainTimeoutSeconds = 5
	srv.Options.Port = 17882
	srv.Options.PortSet = true
	cfg := &configmodel.ConfigModel{LiveKitURL: url, LiveKitAPIKey: localKey, LiveKitAPISecret: localSecret, NativeInProcess: true, NativeLibrary: lib}
	var echoes [2]*acceptanceEcho
	var pubs [2]*processPublisher
	var rooms [2]string
	for i := 0; i < 2; i++ {
		hz, other := 600.0, 1000.0
		if i == 1 {
			hz, other = other, hz
		}
		rooms[i] = fmt.Sprintf("downstream-local-%d-%d", i, time.Now().UnixNano())
		pubs[i] = startProcessPublisher(t, url, rooms[i], hz, other)
		echoes[i] = &acceptanceEcho{hz: hz, other: other}
	}
	jobs := make(chan *worker.JobContext, 2)
	var completed atomic.Int64
	err = srv.RTCSession(func(job *worker.JobContext) error {
		i := 0
		if job.Job.Room.Name == rooms[1] {
			i = 1
		}
		session := agent.NewAgentSession(agent.NewAgent("synthetic echo"), nil, agent.AgentSessionOptions{})
		session.Assistant = echoes[i]
		job.SetSessionDirectory(filepath.Join(out, fmt.Sprint(i)))
		job.InitRecording(agent.RecordingOptions{Audio: true})
		native := newNativeTransport(job, session, cfg, lib, nil)
		if err := job.StartSession(context.Background(), session, worker.StartSessionOptions{Transport: native, RoomOptions: worker.RoomOptions{DisablePreConnectAudio: true, ParticipantIdentity: "caller", RecordRemoteParticipants: true}}); err != nil {
			t.Error(err)
			return err
		}
		hold := sm.NewNativeHoldAudioPort(native, nil)
		session.AddPreCloseCallback(func() {
			if err := hold.Close(); err != nil {
				t.Error(err)
			}
		})
		if i == 1 && os.Getenv("NATIVE_ACCEPTANCE_HOLD") == "1" {
			if _, err := hold.StartHold(context.Background(), rooms[i], holdmodel.HoldAudioConfigModel{Volume: .5}); err != nil {
				t.Error(err)
				return err
			}
		}
		jobs <- job
		<-job.ShutdownDone()
		completed.Add(1)
		return nil
	}, nil, nil)
	if err != nil {
		t.Fatal(err)
	}
	var jobWG sync.WaitGroup
	finished := make(chan struct{})
	srv.OnWorkerStarted(func() {
		for _, room := range rooms {
			jobWG.Add(1)
			go func(name string) {
				defer jobWG.Done()
				if err := srv.ExecuteLocalJobWithOptions(context.Background(), name, "native-agent", worker.LocalJobOptions{FakeJob: false, RoomInfo: &lkproto.Room{Name: name}}); err != nil {
					t.Error(err)
				}
			}(room)
		}
	})
	go func() {
		defer close(finished)
		active := []*worker.JobContext{}
		for len(active) < 2 {
			select {
			case j := <-jobs:
				active = append(active, j)
			case <-time.After(25 * time.Second):
				t.Error("active call startup timeout")
				srv.Close(context.Background())
				syscall.Kill(os.Getpid(), syscall.SIGTERM)
				return
			}
		}
		duration := 120 * time.Second
		if os.Getenv("NATIVE_ACCEPTANCE_HOLD") == "1" || os.Getenv("NATIVE_ACCEPTANCE_FORCE") == "1" {
			duration = 10 * time.Second
		}
		tick := time.NewTicker(10 * time.Second)
		defer tick.Stop()
		until := time.NewTimer(duration)
		defer until.Stop()
	soak:
		for {
			select {
			case <-tick.C:
				var m runtime.MemStats
				runtime.ReadMemStats(&m)
				emit(map[string]any{"event": "sample", "goroutines": runtime.NumGoroutine(), "heap_alloc": m.HeapAlloc, "frames_a": echoes[0].frames.Load(), "frames_b": echoes[1].frames.Load(), "markers_a": pubs[0].markers.Load(), "markers_b": pubs[1].markers.Load()})
			case <-until.C:
				break soak
			}
		}
		for i := range echoes {
			if echoes[i].frames.Load() < 100 || echoes[i].wrong.Load() != 0 || pubs[i].markers.Load() < 50 {
				t.Error("missing or crossed audio", i, echoes[i].frames.Load(), echoes[i].wrong.Load(), pubs[i].markers.Load())
			}
		}
		before := echoes[0].frames.Load()
		at := time.Now()
		if err := syscall.Kill(os.Getpid(), syscall.SIGTERM); err != nil {
			t.Error(err)
			return
		}
		deadline := time.After(10 * time.Second)
		for !srv.Draining() {
			select {
			case <-deadline:
				t.Error("SIGTERM did not enter drain")
				return
			case <-time.After(10 * time.Millisecond):
			}
		}
		time.Sleep(2 * time.Second)
		if echoes[0].frames.Load()-before < 50 {
			t.Error("audio stopped during drain grace")
		}
		if os.Getenv("NATIVE_ACCEPTANCE_FORCE") != "1" {
			for _, job := range active {
				job.Shutdown("local acceptance complete")
			}
		}
		jobWG.Wait()
		emit(map[string]any{"event": "drained", "elapsed_ms": float64(time.Since(at)) / float64(time.Millisecond), "completed_jobs": completed.Load(), "frames_during_grace": echoes[0].frames.Load() - before, "active_jobs": len(srv.ActiveJobs())})
	}()
	oldArgs := os.Args
	os.Args = []string{"worker", "start"}
	defer func() { os.Args = oldArgs }()
	if err := cli.RunAppWithOptionsError(srv); err != nil {
		t.Error(err)
	}
	<-finished
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := closeNativeTransport(ctx); err != nil {
		t.Error(err)
	}
	for _, p := range pubs {
		p.finish()
	}
	emit(map[string]any{"event": "runtime_disposed", "completed_jobs": completed.Load(), "goroutines": runtime.NumGoroutine()})
	if completed.Load() != 2 {
		t.Error("jobs not completed")
	}
}

func processAmplitude(pcm []int16, rate int, hz float64) float64 {
	var re, im float64
	for i, v := range pcm {
		a := 2 * math.Pi * hz * float64(i) / float64(rate)
		re += float64(v) * math.Cos(a)
		im += float64(v) * math.Sin(a)
	}
	return 2 * math.Hypot(re, im) / float64(len(pcm))
}

type processPublisher struct {
	room                 *lk.Room
	stop                 context.CancelFunc
	sent, markers, wrong atomic.Int64
	done                 chan error
	readers              sync.WaitGroup
	err                  chan error
	finish               func()
}

var soakCallback func(*lk.RoomCallback)

func startProcessPublisher(t *testing.T, url, name string, hz, other float64) *processPublisher {
	t.Helper()
	ctx, cancel := context.WithCancel(context.Background())
	p := &processPublisher{stop: cancel, done: make(chan error, 1), err: make(chan error, 8)}
	cb := lk.NewRoomCallback()
	cb.OnTrackSubscribed = func(track *webrtc.TrackRemote, _ *lk.RemoteTrackPublication, _ *lk.RemoteParticipant) {
		p.readers.Add(1)
		go func() {
			defer p.readers.Done()
			decoder, e := opus.NewDecoder(48000, 1)
			if e != nil {
				p.err <- e
				return
			}
			pcm := make([]int16, 5760)
			for {
				packet, _, e := track.ReadRTP()
				if e != nil {
					return
				}
				n, e := decoder.Decode(packet.Payload, pcm)
				if e != nil {
					p.err <- e
					return
				}
				own, alien := processAmplitude(pcm[:n], 48000, hz), processAmplitude(pcm[:n], 48000, other)
				if own > 1000 && own > alien*3 {
					p.markers.Add(1)
				}
				if alien > 1000 && alien > own*3 {
					p.wrong.Add(1)
				}
			}
		}()
	}
	if soakCallback != nil {
		soakCallback(cb)
	}
	var e error
	p.room, e = lk.ConnectToRoom(url, lk.ConnectInfo{APIKey: localKey, APISecret: localSecret, RoomName: name, ParticipantIdentity: "caller"}, cb)
	if e != nil {
		cancel()
		t.Fatal(e)
	}
	track, e := lk.NewLocalTrack(webrtc.RTPCodecCapability{MimeType: webrtc.MimeTypeOpus, ClockRate: 48000, Channels: 2})
	if e != nil {
		t.Fatal(e)
	}
	_, e = p.room.LocalParticipant.PublishTrack(track, &lk.TrackPublicationOptions{Name: "controlled-tone", Source: lkproto.TrackSource_MICROPHONE})
	if e != nil {
		t.Fatal(e)
	}
	go func() {
		enc, e := opus.NewEncoder(48000, 1, opus.AppVoIP)
		if e != nil {
			p.done <- e
			return
		}
		pcm := make([]int16, 960)
		for i := range pcm {
			pcm[i] = int16(10000 * math.Sin(2*math.Pi*hz*float64(i)/48000))
		}
		buf := make([]byte, 4000)
		ticker := time.NewTicker(20 * time.Millisecond)
		defer ticker.Stop()
		var seq uint16
		var ts uint32
		for {
			select {
			case <-ctx.Done():
				p.done <- nil
				return
			case <-ticker.C:
				n, e := enc.Encode(pcm, buf)
				if e == nil {
					e = track.WriteRTP(&rtp.Packet{Header: rtp.Header{Version: 2, SequenceNumber: seq, Timestamp: ts}, Payload: append([]byte(nil), buf[:n]...)}, nil)
				}
				if e != nil {
					p.done <- e
					return
				}
				seq++
				ts += 960
				p.sent.Add(1)
			}
		}
	}()
	p.finish = sync.OnceFunc(func() {
		p.stop()
		p.room.Disconnect()
		select {
		case e := <-p.done:
			if e != nil {
				t.Error(e)
			}
		case <-time.After(3 * time.Second):
			t.Error("publisher did not stop")
		}
		p.readers.Wait()
	})
	t.Cleanup(p.finish)
	return p
}
